// Відображення меню для баристи (із кнопкою видалення) function renderBaristaList() { const menu = getMenu(); const list = document.getElementById('barista-list'); list.innerHTML = ''; menu.forEach((item, index) => { const li = document.createElement('li'); li.innerHTML = ${item.name} (${item.price} грн) ; list.appendChild(li); }); } // Перевірка пароля function checkPassword() { const input = document.getElementById('pass-input').value; if (input === SECRET_PASS) { document.getElementById('login-box').classList.add('hidden'); document.getElementById('admin-box').classList.remove('hidden'); renderBaristaList(); } else { document.getElementById('error-msg').style.display = 'block'; } } // Додавання кави function addNewCoffee() { const name = document.getElementById('coffee-name').value; const price = document.getElementById('coffee-price').value; if (name && price) { const menu = getMenu(); menu.push({ name, price: Number(price) }); saveMenu(menu); document.getElementById('coffee-name').value = ''; document.getElementById('coffee-price').value = ''; } } // Видалення кави function deleteCoffee(index) { const menu = getMenu(); menu.splice(index, 1); saveMenu(menu); } // Первинне завантаження renderMenu();
